// Use an actual CONNECT tunnel. The dependency's historical fixture returns
// from a net/http CONNECT handler without hijacking the connection.
func TestGorillaRealConnectProxyControl(t *testing.T) {
	for _, withAuth := range []bool{false, true} {
		name := map[bool]string{false:"plain",true:"basic-auth"}[withAuth]
		t.Run(name,func(t *testing.T) {
			upgrader := websocket.Upgrader{}
			upstream := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request) {
				c,err := upgrader.Upgrade(w,r,nil)
				if err != nil { return }
				defer c.Close()
				_ = c.SetReadDeadline(time.Now().Add(3*time.Second))
				typ,body,err := c.ReadMessage()
				if err == nil { _ = c.WriteMessage(typ,body) }
			}))
			defer upstream.Close()
			finished := make(chan struct{},1)
			proxy := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request) {
				if r.Method != http.MethodConnect { http.Error(w,"CONNECT required",405);return }
				if withAuth && r.Header.Get("Proxy-Authorization") != "Basic "+base64.StdEncoding.EncodeToString([]byte("synthetic-user:synthetic-password")) { http.Error(w,"auth required",407);return }
				remote,err := net.DialTimeout("tcp",r.Host,time.Second)
				if err != nil { http.Error(w,err.Error(),502);return }
				defer remote.Close()
				client,buffer,err := w.(http.Hijacker).Hijack()
				if err != nil { return }
				defer client.Close()
				defer func(){ finished <- struct{}{} }()
				_,_ = buffer.WriteString("HTTP/1.1 200 Connection Established\r\n\r\n")
				if buffer.Flush()!=nil { return }
				done := make(chan struct{})
				go func(){ _,_ = io.Copy(remote,buffer);_ = remote.Close();close(done) }()
				_,_ = io.Copy(client,remote)
				_ = client.Close()
				<-done
			}))
			defer proxy.Close()
			proxyURL,_ := url.Parse(proxy.URL)
			if withAuth { proxyURL.User = url.UserPassword("synthetic-user","synthetic-password") }
			dialer := websocket.Dialer{Proxy:http.ProxyURL(proxyURL),HandshakeTimeout:time.Second}
			c,_,err := dialer.Dial("ws"+strings.TrimPrefix(upstream.URL,"http"),nil)
			if err != nil { t.Fatal(err) }
			defer c.Close()
			if err := c.WriteMessage(websocket.TextMessage,[]byte("synthetic-loopback"));err != nil { t.Fatal(err) }
			_ = c.SetReadDeadline(time.Now().Add(time.Second))
			_,body,err := c.ReadMessage()
			if err != nil || string(body)!="synthetic-loopback" { t.Fatalf("echo = %q, err = %v",body,err) }
			_ = c.Close()
			select {case <-finished: case <-time.After(2*time.Second):t.Fatal("tunnel did not release")}
		})
	}
}

func TestGorillaSingleAdminSubprotocolControl(t *testing.T) {
	u := websocket.Upgrader{Subprotocols:[]string{"sub2api-admin"}}
	s := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request) {
		c,err := u.Upgrade(w,r,nil)
		if err == nil { _ = c.Close() }
	}))
	defer s.Close()
	d := websocket.Dialer{Subprotocols:[]string{"jwt.synthetic-token","sub2api-admin"},HandshakeTimeout:time.Second}
	c,_,err := d.Dial("ws"+strings.TrimPrefix(s.URL,"http"),nil)
	if err != nil { t.Fatal(err) }
	defer c.Close()
	if got := c.Subprotocol();got!="sub2api-admin" { t.Fatalf("negotiated protocol = %q",got) }
}
