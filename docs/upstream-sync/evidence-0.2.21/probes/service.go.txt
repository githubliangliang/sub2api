func (r *intake021Remote) FetchHashText(context.Context,string)(string,error){return r.hash,nil}
func TestIntake021_PricingHash(t *testing.T){
 old:=[]byte(`{"model-test":{"input_cost_per_token":0.000001,"output_cost_per_token":0.000002}}`)
 next:=[]byte(`{"model-test":{"input_cost_per_token":0.000003,"output_cost_per_token":0.000006}}`)
 r:=&intake021Remote{body:old,hash:fmt.Sprintf("%x",sha256.Sum256(next))}
 s:=NewPricingService(&config.Config{Pricing:config.PricingConfig{DataDir:t.TempDir(),RemoteURL:"https://example.com/pricing.json",HashURL:"https://example.com/pricing.sha256"}},r)
 require.NoError(t,s.ForceUpdate()); r.body=next
 require.NoError(t,s.syncWithRemote())
 require.Equal(t,2,r.downloads)
 require.InDelta(t,3e-6,s.GetModelPricing("model-test").InputCostPerToken,1e-12)
}

func TestIntake021_FastFreeInputControl(t *testing.T){
 s:=&BillingService{fallbackPrices:make(map[string]*ModelPricing)};s.initFallbackPricing()
 zero:=0.0; price:=&ChannelModelPricing{InputPrice:&zero}
 ordinary,e:=s.calculateCostInternal("gpt-6.1-terra",UsageTokens{InputTokens:100,OutputTokens:10},1,"",price);require.NoError(t,e)
 fast,e:=s.calculateCostInternal("gpt-6.1-terra",UsageTokens{InputTokens:100,OutputTokens:10},1,"priority",price);require.NoError(t,e)
 require.Zero(t,fast.InputCost);require.InDelta(t,2*ordinary.OutputCost,fast.OutputCost,1e-12)
}
