func backlogGeminiContext(body string) (*gin.Context,*httptest.ResponseRecorder,*http.Response) {

 return c,rec,&http.Response{StatusCode:http.StatusOK,Header:http.Header{"Content-Type":[]string{"application/json"}},Body:io.NopCloser(strings.NewReader(body))}
}

  body:=[]byte(`{"model":"gpt-5.5","stream":false,"instructions":"fixture","input":[{"role":"user","content":"hello","internal_chat_message_metadata_passthrough":{"content_item_kinds":["text"],"keep":9007199254740993}}]}`)
