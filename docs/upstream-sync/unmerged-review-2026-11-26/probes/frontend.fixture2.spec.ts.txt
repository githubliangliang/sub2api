function deferred<T>() {let resolve!:(v:T)=>void;let reject!:(e:Error)=>void;const promise=new Promise<T>((a,b)=>{resolve=a;reject=b});return {promise,resolve,reject}}
const Dialog={props:['show'],template:'<div v-if="show"><slot/><slot name="footer"/></div>'}

  await w.get('input[type="number"]').setValue('12.5')
  expect((w.emitted('update:modelValue')!.at(-1)![0] as Record<number,unknown>)[1]).toBe('12.5')
 })

  const w=mount(UserAttributeForm,{props:{modelValue:{}},global});await flushPromises();await w.get('input').setValue('text')
  expect((w.emitted('update:modelValue')!.at(-1)![0] as Record<number,unknown>)[1]).toBe('text')

 it('desired: an old user response cannot replace the currently selected user',async()=>{

  expect((w.get('input').element as HTMLInputElement).value).toBe('current-user')
 })
 it('desired: a user response completing after unmount does not emit edited values',async()=>{
  const old=deferred<any[]>();mocks.values.mockReturnValue(old.promise)
  const w=mount(UserAttributeForm,{props:{userId:1,modelValue:{}},global});await flushPromises();w.unmount();old.resolve([{attribute_id:1,value:'late-user'}]);await flushPromises()

 it('desired: switching template ignores old monitor rows and applies only current IDs',async()=>{

describe('shared image upload read ownership',()=>{
 class ControlledReader {
  static readers:ControlledReader[]=[];onload:((e:any)=>void)|null=null;onerror:((e:any)=>void)|null=null;aborted=false
  constructor(){ControlledReader.readers.push(this)} readAsDataURL(){} readAsText(){} abort(){this.aborted=true}
  finish(value:string){if(!this.aborted)this.onload?.({target:{result:value}})}
 }
 beforeEach(()=>{ControlledReader.readers=[];vi.stubGlobal('FileReader',ControlledReader)})
 afterEach(()=>vi.unstubAllGlobals())
 async function choose(w:any,name:string){const input=w.get('input[type="file"]');Object.defineProperty(input.element,'files',{configurable:true,value:[new File(['fixture'],name,{type:'image/png'})]});await input.trigger('change')}
 it('desired: choosing a second image prevents the first read from overwriting it',async()=>{
  const w=shallowMount(ImageUpload,{props:{modelValue:''},global});await choose(w,'first.png');await choose(w,'second.png')
  ControlledReader.readers[1]!.finish('data:image/png;base64,new');ControlledReader.readers[0]!.finish('data:image/png;base64,old');await nextTick()
  expect(w.emitted('update:modelValue')!.at(-1)![0]).toBe('data:image/png;base64,new')
 })
 it('desired: removing an image cancels a pending read',async()=>{
  const w=shallowMount(ImageUpload,{props:{modelValue:'existing'},global});await choose(w,'new.png');await w.get('button').trigger('click');ControlledReader.readers[0]!.finish('late-image');await nextTick()
  expect(w.emitted('update:modelValue')!.at(-1)![0]).toBe('')
 })
 it('control: a single completed read emits the selected image',async()=>{
  const w=shallowMount(ImageUpload,{props:{modelValue:''},global});await choose(w,'one.png');ControlledReader.readers[0]!.finish('selected-image');await nextTick()
  expect(w.emitted('update:modelValue')!.at(-1)![0]).toBe('selected-image')
 })
})
