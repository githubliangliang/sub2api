func TestIntake024RequestScalarOwnsStorage(t *testing.T){
 b:=[]byte(`{"model":"gpt-6.2-sol","input":"`+strings.Repeat("x",1<<20)+`"}`);v:=newOpenAIRequestView(b)
 pos:=uintptr(unsafe.Pointer(unsafe.StringData(v.Model)));start:=uintptr(unsafe.Pointer(unsafe.SliceData(b)))
 require.False(t,pos>=start&&pos<start+uintptr(len(b)),"queued scalar aliases the full 1MiB request buffer")
}

func TestIntake024BufferedHostedCall(t *testing.T){
 acc:=apicompat.NewBufferedResponseAccumulator()
 var ev apicompat.ResponsesStreamEvent
 require.NoError(t,json.Unmarshal([]byte(`{"type":"response.output_item.done","output_index":0,"item":{"type":"web_search_call","id":"ws1","status":"completed","action":{"type":"search","query":"news"}}}`),&ev))
 acc.ProcessEvent(&ev);out,ok:=buildResponsesOutputJSON(acc,nil);require.True(t,ok);require.Equal(t,"web_search_call",gjson.GetBytes(out,"0.type").String())
}
