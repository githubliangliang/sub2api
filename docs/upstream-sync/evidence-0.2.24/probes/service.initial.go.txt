func TestIntake024DuplicateModelParser(t *testing.T) {
 body:=[]byte(`{"model":"gpt-6.2-sol","model":"gpt-6.2-astra","input":"hello"}`)
 parsed,err:=ParseGatewayRequest(NewRequestBodyRef(body),"responses")
 require.NoError(t,err)
 var decoded map[string]any;require.NoError(t,json.Unmarshal(body,&decoded))
 require.Equal(t,"gpt-6.2-sol",parsed.Model)
 require.Equal(t,"gpt-6.2-astra",decoded["model"])
 t.Log("Confirmed divergent model consumers on the unchanged baseline")
}

func TestIntake024OpenAIRotationSurvivesEnrichmentDeadline(t *testing.T){
 a:=&Account{ID:85,Platform:PlatformOpenAI,Type:AccountTypeOAuth,Status:StatusActive,Credentials:map[string]any{"access_token":"old-access","refresh_token":"old-refresh"}}
 repo:=&refreshAPIAccountRepo{account:a}
 e:=&refreshAPIExecutorStub{needsRefresh:true,credentials:map[string]any{"access_token":"rotated-access","refresh_token":"rotated-refresh"},delay:30*time.Millisecond}
 ctx,cancel:=context.WithTimeout(context.Background(),10*time.Millisecond);defer cancel()
 result,err:=NewOAuthRefreshAPI(repo,nil).RefreshIfNeeded(ctx,a,e,time.Hour)
 require.NoError(t,err,"irreversible OpenAI rotation should be saved after optional enrichment expires")
 require.NotNil(t,result);require.Equal(t,1,repo.updateCredentialsCalls)
}
