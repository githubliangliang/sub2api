// These assertions describe the frozen baseline. PASS can prove a defect or a
// negative control; it never means that a proposed release fix is implemented.
func TestRelease019ProbeClaudeUTF16(t *testing.T) {
	const version = "2.1.283"
	for _, sample := range []struct{ text, want string }{
		{"这是一个用于验证中文指纹计算的测试消息，请保持一致。", "c7e"},
		{"你好世界呀", "4ed"},
		{"abcd😀😀efghijklmnopqrstuvwxyz", "0e9"},
	} {
		body, err := json.Marshal(map[string]any{"messages": []map[string]any{{"role": "user", "content": sample.text}}})
		if err != nil { t.Fatal(err) }
		got := computeClaudeCodeFingerprint(body, version)
		if got == sample.want { t.Fatalf("expected baseline byte-index defect, got matching oracle %s", got) }
		t.Logf("DEFECT: Go byte-index fingerprint=%s, independent JS UTF-16 oracle=%s", got, sample.want)
	}
	if got := computeClaudeCodeFingerprint([]byte(`{"messages":[{"role":"user","content":"abcdefghijklmnopqrstuvwxyz"}]}`), version); got != "2e9" {
		t.Fatalf("ASCII control changed: %s", got)
	}
}

func TestRelease019ProbeCodexRefreshAfterFreshFill(t *testing.T) {
	var calls atomic.Int32
	upstream := &codexModelsHTTPUpstreamStub{do: func(_ *http.Request, _ string, _ int64, _ int) (*http.Response, error) {
		calls.Add(1)
		return &http.Response{StatusCode: http.StatusOK, Header: make(http.Header), Body: io.NopCloser(strings.NewReader(`{"models":[]}`))}, nil
	}}
	svc := newCodexModelsAPIKeyTestService(upstream)
	request := codexModelsManifestRequest{url: "https://upstream.example/v1/models", headers: make(http.Header), accountID: 19, useAPIKeyUpstream: true}
	key := buildCodexModelsManifestCacheKey(request)
	_, first := svc.codexModelsManifestCache.get(key, time.Now())
	if first == codexModelsManifestCacheFresh { t.Fatal("initial cache unexpectedly fresh") }
	// Deterministic interleaving: caller observed a miss, another completed flight
	// then filled the cache, and the original caller now enters refresh.
	svc.codexModelsManifestCache.set(key, &CodexModelsManifest{Body: []byte(`{"models":[]}`)}, time.Now())
	result := <-svc.refreshCachedAPIKeyCodexModelsManifest(key, request)
	if result.Err != nil { t.Fatal(result.Err) }
	if calls.Load() != 1 { t.Fatalf("expected one unnecessary baseline fetch, got %d", calls.Load()) }
	t.Log("DEFECT: fresh cache at flight entry still caused one upstream fetch; intended count is zero")
}

func TestRelease019ProbeMultipartTruncation(t *testing.T) {
	var body bytes.Buffer
	writer := multipart.NewWriter(&body)
	if err := writer.WriteField("model", "gpt-image-1"); err != nil { t.Fatal(err) }
	part, err := writer.CreateFormFile("image", "input.png")
	if err != nil { t.Fatal(err) }
	if _, err := io.Copy(part, io.LimitReader(strings.NewReader(strings.Repeat("a", openAIImageMaxUploadPartSize+1)), int64(openAIImageMaxUploadPartSize+1))); err != nil { t.Fatal(err) }
	if err := writer.Close(); err != nil { t.Fatal(err) }
	request := &OpenAIImagesRequest{}
	if err := parseOpenAIImagesMultipartRequest(body.Bytes(), writer.FormDataContentType(), request); err != nil { t.Fatalf("baseline rejected instead of truncating: %v", err) }
	if len(request.Uploads) != 1 { t.Fatalf("unexpected upload count %d", len(request.Uploads)) }
	if len(request.Uploads[0].Data) != openAIImageMaxUploadPartSize { t.Fatalf("unexpected upload length %d", len(request.Uploads[0].Data)) }
	t.Logf("DEFECT: %d-byte upload accepted and truncated to %d bytes", openAIImageMaxUploadPartSize+1, len(request.Uploads[0].Data))
}
