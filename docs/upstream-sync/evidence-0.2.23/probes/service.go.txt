func TestIntake023CountTokensHeaderDuplicates(t *testing.T) {
	c, _ := intake023Context()
	c.Request.Header = http.Header{"content-type": {"application/custom-json"}, "anthropic-version": {"2026-05-01"}}
	account := &Account{ID: 1, Platform: PlatformAnthropic, Type: AccountTypeAPIKey}
	req, err := (&GatewayService{cfg: &config.Config{}}).buildCountTokensRequestAnthropicAPIKeyPassthrough(context.Background(), c, account, []byte("{\"model\":\"claude-sonnet-4-6\",\"messages\":[]}"), "local-fixture-key")
	require.NoError(t, err)
	for _, name := range []string{"content-type", "anthropic-version"} {
		count := 0
		for k, values := range req.Header {
			if strings.EqualFold(k, name) {
				count += len(values)
			}
		}
		require.Equal(t, 1, count, name)
	}
}

func TestIntake023SpoolUnavailable(t *testing.T) {
	stage := newOpenAIFirstOutputStage(128 * 1024)
	stage.createTemp = func() (*os.File, error) { return nil, os.ErrPermission }
	defer stage.Close()
	data := bytes.Repeat([]byte("x"), openAIFirstOutputStageMemoryLimit+1)
	_, err := stage.Write(data)
	require.NoError(t, err, "unavailable spool must retain the same bounded memory fallback")
	var out bytes.Buffer
	require.NoError(t, stage.CommitTo(&out))
	require.Equal(t, data, out.Bytes())
}

func TestIntake023WeeklySurvivesFiveHourRollover(t *testing.T) {
	now := time.Now()
	oldEnd := now.Add(-time.Minute)
	extra := map[string]any{"passive_usage_7d_utilization": 0.9, "passive_usage_7d_reset": now.Add(24 * time.Hour).Unix(), "passive_usage_sampled_at": now.Format(time.RFC3339)}
	repo := &intake023WindowRepo{extra: extra}
	account := &Account{ID: 1, Platform: PlatformAnthropic, Type: AccountTypeSetupToken, SessionWindowEnd: &oldEnd, Extra: extra}
	headers := http.Header{"Anthropic-Ratelimit-Unified-5h-Status": {"allowed"}, "Anthropic-Ratelimit-Unified-5h-Reset": {strconv.FormatInt(now.Add(5*time.Hour).Unix(), 10)}}
	(&RateLimitService{accountRepo: repo}).UpdateSessionWindow(context.Background(), account, headers)
	require.Equal(t, 0.9, repo.extra["passive_usage_7d_utilization"])
}

func TestIntake023DeepSeekOverrideControl(t *testing.T) {
	for _, inputPrice := range []float64{0, 7e-6} {
		cfg := &config.Config{}
		cfg.Pricing.OverrideFile = filepath.Join(t.TempDir(), "override.json")
		patch, err := json.Marshal(map[string]any{"deepseek-chat": map[string]any{
			"input_cost_per_token": inputPrice,
			"output_cost_per_token": 8e-6,
			"cache_read_input_token_cost": 0,
		}})
		require.NoError(t, err)
		require.NoError(t, os.WriteFile(cfg.Pricing.OverrideFile, patch, 0600))
		pricing := &PricingService{cfg: cfg}
		data, err := pricing.parsePricingData([]byte("{\"deepseek-chat\":{\"input_cost_per_token\":1e-6,\"output_cost_per_token\":2e-6,\"cache_read_input_token_cost\":1e-7}}"))
		require.NoError(t, err)
		pricing.pricingData = data
		card, err := NewBillingService(cfg, pricing).GetModelPricing("deepseek-chat")
		require.NoError(t, err)
		require.Equal(t, inputPrice, card.InputPricePerToken)
		require.Equal(t, 8e-6, card.OutputPricePerToken)
		require.Zero(t, card.CacheReadPricePerToken)
	}
}
